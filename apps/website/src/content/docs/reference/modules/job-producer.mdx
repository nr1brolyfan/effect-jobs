## JobProducer

Immutable operation declaration and typed production methods; no transaction ownership on joined enqueue.

```ts title="Type declaration"
export interface JobProducer<Operation extends string, D extends Bindings>
  extends Pipeable.Pipeable {
  readonly operation: Operation
  readonly slots: D
  /** Joins the configured client's active transaction. Providing JobEnqueue alone
   * does not open one. The enqueue result is provisional until the outer commit. */
  readonly enqueue: <const K extends keyof D & string>(
    input: Input<D, K> & {
      readonly slot: K
    }
  ) => Effect.Effect<
    EnqueueResult,
    JobEnqueueError | JobPersistenceError,
    JobEnqueue | D[K]["job"]["payload"]["EncodingServices"]
  >
  /** For a job without an accompanying business write. Requires JobEnqueue and
   * rejects an active transaction on the configured client. Success acknowledges
   * the standalone commit. */
  readonly enqueueStandalone: <const K extends keyof D & string>(
    input: Input<D, K> & {
      readonly slot: K
    }
  ) => Effect.Effect<
    EnqueueResult,
    JobEnqueueError | JobPersistenceError,
    JobEnqueue | D[K]["job"]["payload"]["EncodingServices"]
  >
  /** Joins a qualified custom adapter's explicit JobsTransaction. The application
   * owns commit and rollback. The result is provisional until that commit. */
  readonly enqueueInTransaction: <E, R, const K extends keyof D & string>(
    tx: JobsTransaction<E, R>,
    input: Input<D, K> & {
      readonly slot: K
    }
  ) => Effect.Effect<
    EnqueueResult,
    E | JobEnqueueError,
    R | D[K]["job"]["payload"]["EncodingServices"]
  >
}
```

## make

Starts an immutable producer declaration. Add consequences with withSlot; an empty declaration has no legal enqueue slot. Construction starts no work.

```ts title="Type declaration"
export declare const make: <const Operation extends string>(
  operation: Operation
) => Effect.Effect<JobProducer<Operation, {}>, JobDeclarationError>
```

## Routing

Exactly one route declaration; selectable queues have no implicit default.

```ts title="Type declaration"
export type Routing =
  | {
      readonly queue: JobQueue
      readonly queues?: never
    }
  | {
      readonly queues: readonly [JobQueue, ...JobQueue[]]
      readonly queue?: never
    }
```

## withSlot

Adds one immutable binding, preserving prior slot types and Schema services. Duplicate names fail in the typed channel, never overwrite. Configuration is copied when the setup Effect executes; omitted policy resolves to defaultPolicy. Explicit choices and matching-duplicate first-write-wins semantics are independent of assembly.

```ts title="Type declaration"
export declare const withSlot: <
  const Slot extends string,
  const Job extends JobDefinition,
  const Route extends Routing
>(
  slot: Slot & NoInfer<SingleKey<Slot>>,
  job: Job & NoInfer<ConcreteJob<Job>>,
  options: Route &
    NoInfer<ConcreteRoute<Route>> & {
      readonly policy?: JobPolicy
      readonly defaultQueue?: never
    }
) => <Operation extends string, D extends Bindings, E, R>(
  effect: Effect.Effect<JobProducer<Operation, D>, E, R> &
    ([Extract<Slot, keyof NoInfer<D>>] extends [never] ? unknown : never)
) => Effect.Effect<
  JobProducer<Operation, D & { readonly [K in Slot]: ResolvedBinding<Job, Route> }>,
  E | JobDeclarationError | InvalidProducerBinding,
  R
>
```
