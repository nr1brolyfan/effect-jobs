## makeStore

Adapt one object's storage for JobStorage. Use the same positive operation response budget (milliseconds) for enqueue and execution. Construction runs no SQL.

export declare const makeStore: (
  native: Storage,
  responseBudgetMillis: number
) => {
  claim: (input: unknown) => Effect.Effect<
    | {
        readonly _tag: "Claimed"
        readonly claim: Store.ClaimedJob

    | {
        readonly _tag: "Empty"
      }
    | {
        readonly _tag: "Unknown"
      },
    JobPolicyIncompatible | Store.JobStoreProtocolError,
    never
  >
  reconcileClaim: (input: unknown) => Effect.Effect<
    | {
        readonly _tag: "InsufficientLease"
        readonly ownership: Lifecycle.JobOwnership

    | {
        readonly _tag: "NotOwned"
      }
    | {
        readonly _tag: "Owned"
        readonly claim: Store.ClaimedJob
      }
    | {
        readonly _tag: "Unknown"
      },
    Store.JobStoreProtocolError,
    never
  >
  release: (
    input: unknown,
    phase: "BeforeExecution"
  ) => Effect.Effect<
    | {
        readonly _tag: "Applied"

    | {
        readonly _tag: "Unknown"
      },

    never
  >

    never
  >
  reconcileFinalization: (input: unknown) => Effect.Effect<
    | {
        readonly _tag: "Applied"

        readonly _tag: "OwnershipLost"

        readonly _tag: "StillOwned"
      }
    | {
        readonly _tag: "Unknown"
      },

    never
  >
  recoverExpired: (
    queueInput: unknown,
    limitInput: unknown
  ) => Effect.Effect<number, Store.JobStoreProtocolError | SqliteStoreUnavailable, never>

## withTransaction

Run domain SQL and enqueue within one native SQLite Durable Object transaction. Open once per operation and keep domain writes synchronous through the callback's write capability. Keep other I/O outside the native transaction. Await the returned Effect and the storage sync barrier before acknowledging success. The application owns receipts and reconciliation.

```ts title="Type declaration"
export declare const withTransaction: <A, E, R>(
  native: TransactionStorage,
  operationResponseBudgetMillis: number,
  body: (
    transaction: JobsTransaction<WriteFailure>,
    write: Write
  ) => Effect.Effect<A, E, R>
) => Effect.Effect<
  A,
  E | SqliteTransactionUnavailable | RollbackFailure,
  Exclude<R, import("effect/Scope").Scope>
>
```
