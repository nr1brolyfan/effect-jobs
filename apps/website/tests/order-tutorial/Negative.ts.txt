import { Effect, Layer } from "effect"
import { ApiDefinitions } from "./apps/api/src/orders/Definitions.js"
import { ConfirmationHandlerLayer } from "./apps/worker/src/orders/jobs/SendConfirmation.js"

export const checks = Effect.gen(function* () {
  const { PlaceOrder } = yield* ApiDefinitions
  // @ts-expect-error the introductory producer has no create-invoice slot
  const invalidSlot = PlaceOrder.enqueue({ slot: "create-invoice", operationId: "submission", payload: { orderId: "order-1" } })
  // @ts-expect-error payload identity must be a string
  const invalidPayload = PlaceOrder.enqueue({ slot: "send-confirmation", operationId: "submission", payload: { orderId: 1 } })
  // @ts-expect-error enqueue still requires JobEnqueue
  const missingProducer: Effect.Effect<unknown, unknown> = PlaceOrder.enqueue({ slot: "send-confirmation", operationId: "submission", payload: { orderId: "order-1" } })
  // @ts-expect-error the handler requires TransactionalEmails
  const missingEmails: Layer.Layer<never> = ConfirmationHandlerLayer
  return { invalidSlot, invalidPayload, missingProducer, missingEmails }
})
