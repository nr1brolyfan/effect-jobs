import { Effect, Layer } from "effect"
import { ReportDefinitions, ReportHandlerLayer, ReportPolicy } from "./ReportWorkflow.js"

export const checks = Effect.gen(function* () {
  const { GenerateReport, RequestReport } = yield* ReportDefinitions
  const identity = yield* RequestReport.identity({ operationId: "r-1", slot: "generate" })
  // @ts-expect-error payload units must be numeric
  const invalidPayload = GenerateReport.enqueue({ producer: identity, payload: { reportId: "r-1", rows: [{ product: "x", units: "3" }] }, policy: ReportPolicy })
  // @ts-expect-error only the declared consequence slot is accepted
  const invalidSlot = RequestReport.identity({ operationId: "r-1", slot: "deliver" })
  // @ts-expect-error installing handlers still requires registry and application storage
  const closed: Layer.Layer<never> = ReportHandlerLayer
  // @ts-expect-error an enqueue effect still requires JobEnqueue
  const missingBackend: Effect.Effect<unknown, unknown> = GenerateReport.enqueue({} as never)
  return { invalidPayload, invalidSlot, closed, missingBackend }
})
