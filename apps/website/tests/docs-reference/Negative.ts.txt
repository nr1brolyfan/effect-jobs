import { Effect, Layer } from "effect"
import { ReportDefinitions, ReportHandlerLayer } from "./ReportWorkflow.js"

export const checks = Effect.gen(function* () {
  const { RequestReport } = yield* ReportDefinitions
  // @ts-expect-error payload units must be numeric
  const invalidPayload = RequestReport.enqueue({ slot: "generate", operationId: "r-1", payload: { reportId: "r-1", rows: [{ product: "x", units: "3" }] } })
  // @ts-expect-error only the declared consequence slot is accepted
  const invalidSlot = RequestReport.enqueue({ slot: "deliver", operationId: "r-1", payload: { reportId: "r-1", rows: [] } })
  // @ts-expect-error installing handlers still requires application storage and handoff
  const closed: Layer.Layer<never> = ReportHandlerLayer
  // @ts-expect-error an enqueue effect still requires JobEnqueue
  const missingProducer: Effect.Effect<unknown, unknown> = RequestReport.enqueue({ slot: "generate", operationId: "r-1", payload: { reportId: "r-1", rows: [] } })
  return { invalidPayload, invalidSlot, closed, missingProducer }
})
