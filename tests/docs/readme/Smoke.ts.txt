// Executed on Node/Bun against exact source and installed fences. Test stores
// prove composition only; separate native PG gates prove durable transactions.
import assert from "node:assert/strict"
import { Cause, Effect, Exit, Layer, Result, Schema } from "effect"
import { TestClock } from "effect/testing"
import { BillingQueue, SendReceipt, IssueInvoice, GeneratePdf } from "./contracts.js"
import * as Fixture from "./Fixture.js"
import { snippet as overview } from "./overview.js"
import { issueInvoice } from "./application.js"
import { snippet as slots } from "./slots.js"
import { snippet as standalone } from "./standalone.js"
import { snippet as encrypted } from "./encrypted.js"
import { snippet as policySnippet } from "./policy.js"
import { snippet as migration } from "./migration.js"
import { ReceiptHandler } from "./handler.js"
import { ReceiptHandler as FailureHandler } from "./failures.js"
import { PrivateReceiptHandler } from "./private-handler.js"
import { jobTables } from "./drizzle.js"
import { jobTables as customTables } from "./indexes.js"
import { GeneratePdf as CloudPdf, CloudInvoiceProducer, BillingWorkerLayer } from "./cloud-contracts.js"
import * as Codec from "effect-jobs/JobPayloadCodec"
import * as Lifecycle from "effect-jobs/JobLifecycle"
import * as Store from "effect-jobs/JobStore"
import * as Registry from "effect-jobs/JobRegistry"
import * as Worker from "effect-jobs/JobWorker"
import * as Runtime from "effect-jobs/JobWorkerRuntime"
import * as Consumer from "effect-jobs/JobConsumer"
import * as Policy from "effect-jobs/JobPolicy"
import { JobEnqueue } from "effect-jobs/JobEnqueue"
import type { PreparedJob } from "effect-jobs/JobContract"
import type { JobEnqueueService } from "effect-jobs/JobEnqueue"
import { JobId } from "effect-jobs/JobId"
import { getTableConfig } from "drizzle-orm/pg-core"
assert.equal(process.versions.bun ?? process.version, process.versions.bun ? "1.4.2" : "v24.15.0")
assert.equal(SendReceipt.catalog.version, 1)
assert.equal(BillingQueue.name, "billing")
assert.equal(GeneratePdf.kind, CloudPdf.kind)
assert.equal(Layer.isLayer(BillingWorkerLayer), true)
assert.equal(CloudInvoiceProducer.identity({ operationId: "op", slot: "generate-pdf" }).slot, "generate-pdf")
assert.equal(IssueInvoice.identity({ operationId: "op", slot: "receipt" }).operation, "billing.issue-invoice")
assert(Object.isFrozen(Policy.defaultPolicy))
assert(jobTables.jobs && jobTables.payloads)
assert(getTableConfig(customTables.jobs).indexes.some((i) => i.config.name === "jobs_dashboard"))
const initial = await Effect.runPromiseExit(overview)
assert(Exit.isFailure(initial), "overview delegates to explicitly supplied app manager; no fallback")
const prepared: PreparedJob[] = []
const record: JobEnqueueService["enqueue"] = (effect) => Effect.flatMap(effect, (value) => Effect.sync(() => {
  prepared.push(value)
  return { _tag: "Inserted" as const, jobId: Schema.decodeSync(JobId)(`readme-job-${prepared.length}`) }
}))
const producer: JobEnqueueService = { enqueue: record, enqueueStandalone: record }
const backend = Fixture.jobsStorage
Fixture.useBackend({ ...backend, producer, withTransaction: (body) => Effect.suspend(() => body({} as never)).pipe(Effect.provideService(JobEnqueue, producer)) })
await Effect.runPromise(issueInvoice("op", "invoice"))
await Effect.runPromise(slots.pipe(Effect.provideService(JobEnqueue, producer)))
assert.deepEqual(prepared.slice(1, 3).map((p) => p.producer.slot), ["send-receipt", "generate-pdf"])
await Effect.runPromise(standalone)
const policy = await Effect.runPromise(policySnippet.pipe(Effect.provideService(JobEnqueue, producer)))
assert.equal(policy.attemptTimeoutMillis, 20000)
assert.equal(policy.maxAttempts, 4)
assert(Exit.isFailure(await Effect.runPromiseExit(migration)), "readiness requires real app SQL, never silently runs DDL")
const ring = Fixture.keys()
const privateDefinitions = await Effect.runPromise(encrypted.pipe(Effect.provide(Fixture.keysLayer(ring))))
const privatePrepared = prepared.at(-1)!
const stored = new TextDecoder().decode(privatePrepared.encoded.payloadBytes)
assert(!stored.includes("alex@example.com"))
assert(!stored.includes("INV-2026-1042"))
const privateValue = { invoiceId: "private-invoice", document: { recipient: "alex@example.com", invoiceNumber: "INV-2026-1042", total: { amount: "49.00", currency: "EUR" } } }
const encode = (keyring: Fixture.Keyring) => Effect.runPromise(Codec.encodeJobPayload(privateDefinitions.SendPrivateReceipt.payload, privateValue).pipe(Effect.provide(Fixture.keysLayer(keyring))))
const a = await encode(ring), b = await encode(ring), rotated = await encode({ ...ring, active: "two" })
assert.notDeepEqual(a.payloadBytes, b.payloadBytes)
assert.deepEqual(a.semanticProjectionBytes, b.semanticProjectionBytes)
assert.deepEqual(a.semanticProjectionBytes, rotated.semanticProjectionBytes)
assert.deepEqual(await Effect.runPromise(Codec.decodeJobPayload(privateDefinitions.SendPrivateReceipt.payload, a).pipe(Effect.provide(Fixture.keysLayer({ ...ring, active: "two" })))), privateValue)
assert(Exit.isFailure(await Effect.runPromiseExit(Codec.encodeJobPayload(privateDefinitions.SendPrivateReceipt.payload, privateValue).pipe(Effect.provide(Fixture.keysLayer({ ...ring, encryption: {} }))))))
assert(Exit.isFailure(await Effect.runPromiseExit(Effect.void.pipe(Effect.provide(Fixture.keysLayer({ ...ring, encryption: {} }))))))
assert(Exit.isFailure(await Effect.runPromiseExit(Codec.decodeJobPayload(privateDefinitions.SendPrivateReceipt.payload, a).pipe(Effect.provide(Fixture.keysLayer({ ...ring, encryption: { one: new Uint8Array(32) } }))))))
const context = { jobId: Schema.decodeSync(JobId)("handler-job"), catalog: SendReceipt.catalog, producer: IssueInvoice.identity({ operationId: "op", slot: "receipt" }), attemptNumber: 1 }
const encoded = await Effect.runPromise(Codec.encodeJobPayload(SendReceipt.payload, { invoiceId: "invoice" }))
let calls = 0
const registry = Registry.layer
await Effect.runPromise(Effect.gen(function* () {
  const r = yield* Registry.JobRegistry
  yield* r.find(SendReceipt.catalog)!.execute(encoded, context)
}).pipe(Effect.provide(Layer.merge(registry, ReceiptHandler.pipe(Layer.provide(registry), Layer.provide(Layer.succeed(Fixture.ReceiptMailer, { send: (input) => Effect.sync(() => { assert.equal(input.idempotencyKey, "handler-job"); calls++ }) })))))))
assert.equal(calls, 1)
for (const [tag, expected] of [["Accepted", null], ["RateLimited", "Retry"], ["InvalidRecipient", "Dead"], ["CredentialsRejected", "Isolate"], ["Unconfirmed", "OutcomeUnknown"]] as const) {
  const layer = FailureHandler.pipe(Layer.provide(registry), Layer.provide(Layer.mergeAll(
    Layer.succeed(Fixture.ReceiptTemplates, { load: () => Effect.succeed("template") }),
    Layer.succeed(Fixture.EmailProvider, { send: () => Effect.succeed({ _tag: tag }) })
  )))
  const exit = await Effect.runPromiseExit(Effect.gen(function* () { const r = yield* Registry.JobRegistry; return yield* r.find(SendReceipt.catalog)!.execute(encoded, context) }).pipe(Effect.provide(Layer.merge(registry, layer))))
  if (expected === null) assert(Exit.isSuccess(exit))
  else { assert(Exit.isFailure(exit)); const reasons = exit.cause.reasons; assert(Cause.isFailReason(reasons[0]!)); assert.equal(reasons[0]!.error._tag, expected) }
}
let privateCalls = 0
await Effect.runPromise(Effect.gen(function* () {
  const r = yield* Registry.JobRegistry
  yield* r.find(Fixture.SendPrivateReceipt.catalog)!.execute(a, { ...context, catalog: Fixture.SendPrivateReceipt.catalog })
}).pipe(Effect.provide(Layer.merge(registry, PrivateReceiptHandler.pipe(Layer.provide(registry), Layer.provide(Fixture.keysLayer(ring)), Layer.provide(Layer.succeed(Fixture.PrivateReceiptMailer, { send: (input) => Effect.sync(() => { assert.deepEqual(input.document, privateValue.document); privateCalls++ }) })))))))
assert.equal(privateCalls, 1)
// A real worker consumes a claimed job and owns finalization. The store is a double.
let claims = 0, finalized = 0
const snapshot = Result.getOrThrow(Lifecycle.claim({ jobId: context.jobId, policy: Policy.defaultPolicy, state: "Pending", availableAt: 100000, updatedAt: 100000, attemptsMade: 0, stalledCount: 0, lifecycleVersion: 0, leaseToken: null, leaseExpiresAt: null, completedAt: null, lastFailureCode: null }, "abcd-1234-feed-5678", 100000, 100))
const ownership = Result.getOrThrow(Lifecycle.ownershipOf(snapshot))
const unexpected = () => Effect.die("Unexpected harness operation")
const store: Store.JobStoreService = {
  claim: (request) => Effect.sync(() => ++claims === 1 ? Store.ClaimResults.Claimed({ claim: { snapshot: { ...snapshot, leaseToken: request.leaseToken }, ownership: { ...ownership, leaseToken: request.leaseToken }, prepared: { catalog: SendReceipt.catalog, producer: context.producer, policy: Policy.defaultPolicy, encoded } } }) : Store.ClaimResults.Empty()),
  recoverExpired: () => Effect.succeed(0), finalize: () => Effect.sync(() => { finalized++; return Store.FinalizationResults.Applied() }), release: unexpected, reconcileClaim: unexpected, reconcileFinalization: unexpected
}
const worker = Worker.layer(store, { catalog: [SendReceipt], operationResponseBudgetMillis: 100 }).pipe(Layer.provideMerge(ReceiptHandler), Layer.provide(registry), Layer.provide(Fixture.ReceiptMailerLayer))
const consumer = await Effect.runPromise(Consumer.make(BillingQueue, { localConcurrency: 2, claimLimitPerRun: 20, recoveryLimitPerRun: 20 }))
const result = await Effect.runPromise(Effect.scoped(Runtime.drain(consumer).pipe(Effect.provide(worker))).pipe(Effect.provide(TestClock.layer())))
assert.equal(result.claimed, 1)
assert.equal(finalized, 1)
console.log("README exact fences: producer slots/service, policy/migration/indexes, handler outcomes, AES-GCM/HMAC typed keys and finite worker PASS")
