// Executed against extracted source and installed snippets, on Node and Bun.
// Test doubles here certify composition only, never durable SQL/atomicity.
import assert from "node:assert/strict"
import { Deferred, Effect, Fiber, Layer, Result, Schema } from "effect"
import { TestClock } from "effect/testing"
import { BillingQueue, SendReceipt, IssueInvoice } from "./shared.js"
import { workerLayers } from "./handler.js"
import { consumerProcess } from "./polling.js"
import { ReceiptMailer } from "./Services.js"
import * as Consumer from "effect-jobs/JobConsumer"
import * as Runtime from "effect-jobs/JobWorkerRuntime"
import { migrationSql, tables } from "./migration.js"
import { jobTables } from "./drizzle.js"
import { policy, rejected, uncertain } from "./policy.js"
import { ProtectedDocument, DocumentPayload, encodeDocument } from "./protection.js"
import * as Codec from "effect-jobs/JobPayloadCodec"
import * as Lifecycle from "effect-jobs/JobLifecycle"
import * as Store from "effect-jobs/JobStore"
import { JobId } from "effect-jobs/JobId"

assert.equal(
  process.versions.bun ?? process.version,
  process.versions.bun ? "1.4.2" : "v24.15.0"
)
assert.equal(BillingQueue.name, "billing")
assert.deepEqual(SendReceipt.catalog, {
  queue: "billing",
  kind: "invoice.receipt-email",
  version: 1
})
assert.deepEqual(
  IssueInvoice.identity({ operationId: "operation-1", slot: "receipt" }),
  {
    operation: "billing.issue-invoice",
    operationId: "operation-1",
    slot: "receipt"
  }
)
assert.equal(tables.schema, "billing")
assert.match(migrationSql, /"billing"\."jobs"/)
assert(jobTables.jobs && jobTables.payloads)
assert.equal(policy.leaseDurationMillis, 90000)
assert.equal(policy.attemptTimeoutMillis, 30000)
assert.equal(policy.maxAttempts, 3)
assert(policy.completedRetention._tag === "Duration")
assert.equal(policy.completedRetention.millis, 90 * 86400000)
assert.equal(policy.deadRetention._tag, "Forever")
assert.equal(Effect.runSync(Effect.flip(rejected))._tag, "Retry")
assert.equal(Effect.runSync(Effect.flip(uncertain))._tag, "OutcomeUnknown")

const document = {
  ciphertext: "test-envelope-a",
  keyId: "test-key",
  fingerprint: "test-fingerprint"
}
assert(Schema.is(ProtectedDocument)(document))
const encoded = await Effect.runPromise(encodeDocument(document))
const rotated = await Effect.runPromise(
  encodeDocument({ ...document, ciphertext: "test-envelope-b" })
)
assert.deepEqual(encoded.semanticProjectionBytes, rotated.semanticProjectionBytes)
assert.notDeepEqual(encoded.payloadBytes, rotated.payloadBytes)
assert.deepEqual(
  await Effect.runPromise(Codec.decodeJobPayload(DocumentPayload, encoded)),
  { document }
)
const changed = await Effect.runPromise(
  encodeDocument({ ...document, fingerprint: "different" })
)
assert.notDeepEqual(encoded.semanticProjectionBytes, changed.semanticProjectionBytes)

// Portable worker harness: actual definition/codec/handler, test-only store.
let executions = 0
let claims = 0
let finalizations = 0
await Effect.runPromise(
  Effect.scoped(
    Effect.gen(function* () {
      const encoded = yield* Codec.encodeJobPayload(SendReceipt.payload, {
        invoiceId: "invoice-1"
      })
      const snapshot = Result.getOrThrow(
        Lifecycle.claim(
          {
            jobId: Schema.decodeSync(JobId)("test-job-1"),
            policy,
            state: "Pending",
            availableAt: 100000,
            updatedAt: 100000,
            attemptsMade: 0,
            stalledCount: 0,
            lifecycleVersion: 0,
            leaseToken: null,
            leaseExpiresAt: null,
            completedAt: null,
            lastFailureCode: null
          },
          "abcd-1234-feed-5678",
          100000,
          100
        )
      )
      const ownership = Result.getOrThrow(Lifecycle.ownershipOf(snapshot))
      const unexpected = () => Effect.die("Unexpected harness operation")
      const store: Store.JobStoreService = {
        claim: (request) =>
          Effect.sync(() => {
            claims++
            return claims === 1
              ? Store.ClaimResults.Claimed({
                  claim: {
                    snapshot: { ...snapshot, leaseToken: request.leaseToken },
                    ownership: { ...ownership, leaseToken: request.leaseToken },
                    prepared: {
                      catalog: SendReceipt.catalog,
                      producer: IssueInvoice.identity({
                        operationId: "operation-1",
                        slot: "receipt"
                      }),
                      policy,
                      encoded
                    }
                  }
                })
              : Store.ClaimResults.Empty()
          }),
        recoverExpired: () => Effect.succeed(0),
        finalize: () =>
          Effect.sync(() => {
            finalizations++
            return Store.FinalizationResults.Applied()
          }),
        reconcileClaim: unexpected,
        release: unexpected,
        reconcileFinalization: unexpected
      }
      const sent = yield* Deferred.make<void>()
      const mailerLayer = Layer.succeed(ReceiptMailer, {
        send: ({ invoiceId, idempotencyKey }) => Effect.sync(() => {
          assert.equal(invoiceId, "invoice-1")
          assert.equal(idempotencyKey, "test-job-1")
          executions++
        }).pipe(Effect.andThen(Deferred.succeed(sent, undefined)))
      })
      const backend = { store, ready: Effect.void }
      const { WorkerLayer } = workerLayers(backend, mailerLayer)
      assert.equal(executions, 0, "Layer construction must not send")
      const consumer = yield* Consumer.make(BillingQueue, {
        localConcurrency: 2, claimLimitPerRun: 20, recoveryLimitPerRun: 20
      })
      const result = yield* Runtime.drain(consumer).pipe(Effect.provide(WorkerLayer))
      assert.equal(result._tag, "Idle")
      assert.equal(result.claimed, 1)
      assert.equal(executions, 1)
      assert.equal(finalizations, 1)

      // Execute the exact README polling program, then interrupt its owning Scope.
      claims = 0
      const polled = yield* Deferred.make<void>()
      const pollingMailerLayer = Layer.succeed(ReceiptMailer, {
        send: ({ invoiceId, idempotencyKey }) => Effect.sync(() => {
          assert.equal(invoiceId, "invoice-1")
          assert.equal(idempotencyKey, "test-job-1")
          executions++
        }).pipe(Effect.andThen(Deferred.succeed(polled, undefined)))
      })
      const application = consumerProcess(backend, pollingMailerLayer)
      assert.equal(executions, 1, "Polling construction must not start work")
      const fiber = yield* application.runWorker.pipe(Effect.forkScoped)
      yield* Deferred.await(polled)
      // Let the handler's fenced finalization and idle sleep settle deterministically.
      for (let i = 0; i < 20; i++) yield* Effect.yieldNow
      assert.equal(finalizations, 2)
      yield* Fiber.interrupt(fiber)
      const stopped = claims
      yield* TestClock.adjust(60000)
      assert.equal(claims, stopped, "Closing the worker Scope must stop polling")
      assert.equal(executions, 2)
    })
  ).pipe(Effect.provide(TestClock.layer()))
)
console.log(
  "README exact fences: declarations, protection, receipt mail adapter, finite drain and scoped polling shutdown PASS (portable store; PostgreSQL not executed)"
)
