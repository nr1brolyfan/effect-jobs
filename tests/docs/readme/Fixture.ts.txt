export const Document = Schema.Struct({ recipient: Schema.String, invoiceNumber: Schema.String, total: Schema.Struct({ amount: Schema.String, currency: Schema.String }) })
export const Envelope = Schema.Struct({ keyId: Schema.String, nonce: Schema.String, ciphertext: Schema.String, tag: Schema.String, fingerprint: Schema.String })
export interface Keyring { readonly active: string; readonly encryption: Readonly<Record<string, Uint8Array>>; readonly fingerprint: Uint8Array }
export class ReceiptKeys extends Context.Service<ReceiptKeys, Keyring>()("readme/ReceiptKeys") {}
export const keys = (): Keyring => ({ active: "one", encryption: { one: randomBytes(32), two: randomBytes(32) }, fingerprint: randomBytes(32) })
export const keysLayer = (ring: Keyring) => Layer.effect(ReceiptKeys, Effect.gen(function* () {
  if (ring.encryption[ring.active]?.byteLength !== 32 || ring.fingerprint.byteLength !== 32) return yield* new Payload.JobEncryptionError()
  return ring
}))
const canonical = (value: typeof Document.Encoded) => JSON.stringify({ recipient: value.recipient, invoiceNumber: value.invoiceNumber, total: { amount: value.total.amount, currency: value.total.currency } })
export const ReceiptEncryption: Payload.EncryptionCodec<typeof Document.Encoded, typeof Envelope, ReceiptKeys, ReceiptKeys> = {
  envelope: Envelope,
  seal: (value) => Effect.flatMap(ReceiptKeys, (ring) => Effect.try({
    try: () => {
      const plain = canonical(value), nonce = randomBytes(12)
      const cipher = createCipheriv("aes-256-gcm", ring.encryption[ring.active]!, nonce)
      const ciphertext = Buffer.concat([cipher.update(plain, "utf8"), cipher.final()])
      return { keyId: ring.active, nonce: nonce.toString("base64"), ciphertext: ciphertext.toString("base64"), tag: cipher.getAuthTag().toString("base64"), fingerprint: createHmac("sha256", ring.fingerprint).update(plain).digest("hex") }
    }, catch: () => new Payload.JobEncryptionError()
  })),
  open: (envelope) => Effect.flatMap(ReceiptKeys, (ring) => Effect.try({
    try: () => {
      const key = ring.encryption[envelope.keyId]
      if (!key) throw new Error("Unknown fixture key")
      const cipher = createDecipheriv("aes-256-gcm", key, Buffer.from(envelope.nonce, "base64"))
      cipher.setAuthTag(Buffer.from(envelope.tag, "base64"))
      const plain = Buffer.concat([cipher.update(Buffer.from(envelope.ciphertext, "base64")), cipher.final()]).toString("utf8")
      if (createHmac("sha256", ring.fingerprint).update(plain).digest("hex") !== envelope.fingerprint) throw new Error("Fingerprint mismatch")
      return JSON.parse(plain) as typeof Document.Encoded
    }, catch: () => new Payload.JobEncryptionError()
  }))
}
